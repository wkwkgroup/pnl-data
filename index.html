  1. Buka sheets.new di browser untuk membuat Google Sheet baru.

  3. Hapus isi kode lama, lalu tempel kode di bawah ini & klik Save.
  4. Klik Deploy → New deployment → Web app, atur Who has access ke Anyone, lalu salin URL /exec ke kolom di atas.
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  var rows = sheet.getDataRange().getValues();
  var entries = [];
  var currency = 'USD';
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      var d = rows[i][0] instanceof Date
        ? Utilities.formatDate(rows[i][0], ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd')
        : String(rows[i][0]).trim();
      entries.push({
        date: d,
        equity: Number(rows[i][1]) || 0,
        netFlow: Number(rows[i][2]) || 0,
        note: rows[i][5] ? String(rows[i][5]) : ''
      });
      if (rows[i][4]) currency = String(rows[i][4]);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    currency: currency,
    entries: entries
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  var body = JSON.parse(e.postData.contents);
  var entries = body.entries || [];
  var currency = body.currency || 'USD';
  sheet.clearContents();
  sheet.appendRow(['Date', 'Equity_USD', 'Net_Flow_USD', 'Updated_At', 'Currency', 'Note']);
  var now = new Date().toISOString();
  for (var i = 0; i < entries.length; i++) {
    sheet.appendRow([
      entries[i].date,
      Number(entries[i].equity) || 0,
      Number(entries[i].netFlow) || 0,
      now,
      currency,
      entries[i].note || ''
    ]);
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    saved: entries.length
  })).setMimeType(ContentService.MimeType.JSON);
}
