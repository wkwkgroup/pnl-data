  1. Buka sheets.new di browser untuk membuat Google Sheet baru.

  3. Hapus isi kode lama, lalu tempel kode di bawah ini & klik Save.
  4. Klik Deploy → New deployment → Web app, atur Who has access ke Anyone, lalu salin URL /exec ke kolom di atas.
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  var rows = sheet.getDataRange().getValues();
  var entries = [];
  var currency = 'IDR';
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      var d = rows[i][0] instanceof Date
        ? Utilities.formatDate(rows[i][0], ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd')
        : String(rows[i][0]).trim();
      entries.push({
        date: d,
        equity: Number(rows[i][1]) || 0,
        netFlow: Number(rows[i][2]) || 0,
        note: rows[i][5] ? String(rows[i][5]) : ''
      });
      if (rows[i][4]) currency = String(rows[i][4]);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    currency: currency,
    entries: entries
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  var payload = JSON.parse(e.postData.contents);
  var entries = payload.entries || [];
  var currency = payload.currency || 'IDR';
  sheet.clear();
  sheet.appendRow(['Date', 'Equity', 'Net_TopUp', 'Updated_At', 'Currency', 'Note', 'VOO', 'SCHD']);
  var now = new Date().toISOString();
  var prevEq = 0;
  for (var i = 0; i < entries.length; i++) {
    var item = entries[i];
    var eq = Number(item.equity) || 0;
    var nf = Number(item.netFlow) || 0;
    var pnlNominal = i === 0 ? 0 : (eq - prevEq - nf);
    var pnlPct = (i === 0 || (prevEq + nf) === 0) ? 0 : (pnlNominal / (prevEq + nf));
    sheet.appendRow([item.date, eq, nf, now, currency, item.note || '', pnlNominal, pnlPct]);
    prevEq = eq;
  }
  if (entries.length > 0) {
    sheet.getRange(2, 7, entries.length, 1).setNumberFormat('#,##0');
    sheet.getRange(2, 8, entries.length, 1).setNumberFormat('+0.00%;-0.00%;0.00%');
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'saved', count: entries.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
