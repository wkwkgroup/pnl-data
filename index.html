function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  var rows = sheet.getDataRange().getValues();
  var entries = [];
  var currency = 'IDR';
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      var d = rows[i][0] instanceof Date
        ? Utilities.formatDate(rows[i][0], ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd')
        : String(rows[i][0]).trim();
      var eq = Number(rows[i][1]) || 0;
      var voo = rows[i][6] !== undefined && rows[i][6] !== '' ? Number(rows[i][6]) : eq;
      var schd = rows[i][7] !== undefined && rows[i][7] !== '' ? Number(rows[i][7]) : 0;
      entries.push({
        date: d,
        equity: eq,
        netFlow: Number(rows[i][2]) || 0,
        note: rows[i][5] ? String(rows[i][5]) : '',
        voo: voo,
        schd: schd
      });
      if (rows[i][4]) currency = String(rows[i][4]);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    currency: currency,
    entries: entries
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('ETF_PnL') || ss.insertSheet('ETF_PnL');
  sheet.clearContents();
  var output = [['Date', 'Equity', 'NetTopUp', 'UpdatedAt', 'Currency', 'Note', 'VOO', 'SCHD']];
  var now = new Date().toISOString();
  var list = payload.entries || [];
  for (var i = 0; i < list.length; i++) {
    var eq = Number(list[i].equity) || 0;
    var voo = list[i].voo !== undefined ? Number(list[i].voo) : eq;
    var schd = list[i].schd !== undefined ? Number(list[i].schd) : 0;
    output.push([
      list[i].date,
      eq,
      Number(list[i].netFlow) || 0,
      i === 0 ? now : '',
      i === 0 ? (payload.currency || 'IDR') : '',
      list[i].note || '',
      voo,
      schd
    ]);
  }
  sheet.getRange(1, 1, output.length, 8).setValues(output);
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    count: list.length
  })).setMimeType(ContentService.MimeType.JSON);
}
